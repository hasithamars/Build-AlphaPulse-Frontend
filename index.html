async function loadYahooCandles(meta, tf) { const tfMap = { '1m': { interval: '1m', range: '1d' }, '5m': { interval: '5m', range: '2d' }, '15m':{ interval: '15m', range: '5d' }, '1h': { interval: '60m', range: '1mo' }, '4h': { interval: '60m', range: '3mo' }, '1d': { interval: '1d', range: '1y' }, }; const c = tfMap[tf] || tfMap['15m']; const raw = `https://query1.finance.yahoo.com/v8/finance/chart/${encodeURIComponent(meta.yahoo)}?interval=${c.interval}&range=${c.range}`; const r = await fetch(proxied(raw), { signal: AbortSignal.timeout(9000) }); if (!r.ok) throw new Error(`Yahoo HTTP ${r.status}`); const j = await r.json(); const result = j.chart?.result?.[0]; if (!result?.timestamp) throw new Error('Yahoo returned no data'); const q = result.indicators.quote[0]; const out = []; result.timestamp.forEach((t, i) => { if (q.close[i] != null && !isNaN(q.close[i])) { out.push({ time: t, open: +(q.open[i] ?? q.close[i]), high: +(q.high[i] ?? q.close[i]), low: +(q.low[i] ?? q.close[i]), close: +q.close[i], }); } }); if (!out.length) throw new Error('Yahoo candle array empty'); state.dataSource = 'Yahoo (worker)'; return out; }
